cmake_minimum_required(VERSION 3.22)
project(littlebit_image LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
foreach(option BUILD_SHARED_LIBS SD_BUILD_EXAMPLES SD_BUILD_SHARED_LIBS SD_BUILD_SHARED_GGML_LIB SD_WEBP SD_WEBM GGML_NATIVE GGML_OPENMP GGML_BACKEND_DL)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(SD_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
set(Vulkan_INCLUDE_DIR "$ENV{VULKAN_INCLUDE_DIR}" CACHE PATH "")
set(Vulkan_GLSLC_EXECUTABLE "$ENV{VULKAN_GLSLC_EXECUTABLE}" CACHE FILEPATH "")
find_library(Vulkan_LIBRARY vulkan REQUIRED)
find_package(SPIRV-Headers CONFIG REQUIRED HINTS "$ENV{VULKAN_SDK}" NO_CMAKE_FIND_ROOT_PATH)
set(SD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/stable-diffusion.cpp")
add_subdirectory("${SD_ROOT}" sd EXCLUDE_FROM_ALL)
add_library(sd_jni SHARED sd_jni.cpp)
target_compile_features(sd_jni PRIVATE cxx_std_17)
target_include_directories(sd_jni PRIVATE "${SD_ROOT}/include")
target_link_libraries(sd_jni PRIVATE stable-diffusion log android)
target_link_options(sd_jni PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
